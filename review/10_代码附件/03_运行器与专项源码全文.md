"""Run one behavior-only job, without overwriting any experiment or archive."""
import argparse
from datetime import datetime
import math
from pathlib import Path

from agents.government_agent import get_farmer_net_income
from scripts.national_actor_trace import write_actor_trace
from scripts.run_national_inertia_pairs import OLD, legacy, old, parcels, snapshot_plot_use


ROOT = old.ROOT / "results/diagnostics/national_household_portfolio"
ARCHIVE = old.ROOT / "results/diagnostics/national_inertia_pairs_20261121/run_02"
POLICY = {"base": 0, "ordinary_post": 1, "scale80_post": 2}
FILES = ("decision_trace.json", "actors.jsonl", "cash_flow.json", "fiscal_account.json",
         "actual_harvests.json", "actual_plot_use.json", "plot_production_status.json",
         "land.json", "lease_transactions.json", "government_sar.json")


def run(seed, rule, policy="base", paired_old=None):
    if seed not in (103, 109) or rule not in ("old", "new") or policy != "base":
        raise ValueError("behavior phase: seeds 103/109 and base only; policy phase locked")
    output = ROOT / "runs" / (f"s{seed}_{rule}_{policy}_" + datetime.now().strftime("%Y%m%d_%H%M%S_%f"))
    archive = ARCHIVE / f"s{seed}_m50_base_open"
    cfg, economics, budget, inputs = legacy.load_base()
    sources = legacy.source_hashes()
    assert inputs == old.read(ARCHIVE / "design.json")["frozen_inputs"]
    portfolio = rule == "new"
    cfg["runtime_fixed_rule"]["household_crop_portfolio"] = portfolio
    share_path = ROOT / "reference/initial_grain_shares.json"
    if portfolio:
        if paired_old is None:
            raise ValueError("supply an audited old run of the same seed")
        manifest = old.read(paired_old / "audit.json")
        assert manifest["ok"] and (manifest["seed"], manifest["rule"]) == (seed, "old")
        shares = old.read(share_path)
    model = legacy.build_model(cfg, economics, seed)
    if portfolio:
        ids = {str(a.unique_id) for a in model.initial_household_cohort}
        assert shares and shares.keys() <= ids
        assert all(type(x) in (float, int) and 0 <= x <= 1 for x in shares.values())
        model.household_initial_grain_shares = shares
    model.config["policy"]["government_q_project"].update(
        episode_end_step=3, uncapped_income_reward=True,
        grain_reference_kg=old.read(OLD / "training.json")["grain_reference_kg"],
        budget_area_menu=math.fsum(p.area for p in model.land_patches),
        h0_income_reference_yuan=math.fsum(get_farmer_net_income(f) for f in model.initial_household_cohort)/600)
    expected_cfg = old.read(archive / "effective_config.json")
    expected_cfg["runtime_fixed_rule"]["household_crop_portfolio"] = portfolio
    assert model.config == expected_cfg
    assert model.config["runtime_fixed_rule"]["minimum_inertia_margin_yuan_per_mu"] == 50
    legacy.configure_government(model.government, {}, seed, False)
    # The inertia experiment fixed the choice directly, without counting unknown Q states.
    model.government._choose_action_rl = lambda state, action_count=3: POLICY[policy]
    output.mkdir(parents=True)
    path = output / "episode"
    old.write(output / "design.json", dict(seed=seed, rule=rule, policy=policy, policy_locked=True,
              source_hashes=sources, runner_sha256=old.digest(Path(__file__)), inputs=inputs,
              initial_parcels_sha256=legacy.fingerprint(parcels(model)), archive=str(archive),
              share_sha256=old.digest(share_path) if portfolio else None))

    def observe(current, year):
        old.write(year / "actual_plot_use.json", snapshot_plot_use(current))
        write_actor_trace(current, year)
        env = current.environment
        cfg = current.config
        last = env.labor_market_history[-1]
        old.write(year / "price_inputs.json", {
            "year": current.simulation_base_year + current.step_count,
            "base_crop_prices": cfg["environment"]["market"]["base_prices"],
            "base_land_rent": cfg["environment"]["market"]["base_land_rent"],
            "other_cost_per_mu": cfg["production"]["cost_structure"]["input_requirements"]["other_cost_per_mu"],
            "labor_shock_factor": last["shock_factor"],
        })
        old.write(year / "market_outcomes.json", dict(env.market_prices))

    print(path, flush=True)
    result = legacy.run_episode(model, budget, path, seed, sources, False, full_audit=True, observe_year=observe)
    assert not result["training"] and result["frozen_q_unchanged"]
    baseline = paired_old / "episode" if portfolio else archive
    checks = {}
    def compare(rel):
        lhs, rhs = old.digest(path / rel), old.digest(baseline / rel)
        checks[rel] = dict(match=lhs == rhs, actual=lhs, reference=rhs)
    for name in ("initial_households.json", "initial_operators.json", "initial_exogenous.json"):
        compare(name)
    for i in (1, 2, 3):
        year = f"year_{i:02d}/"
        prices = old.read(path / (year+"price_inputs.json"))
        scenario = expected_cfg["paper_b_flagship_scenario"]
        assert prices["year"] == 2010+i
        assert prices["base_crop_prices"] == expected_cfg["environment"]["market"]["base_prices"]
        assert prices["base_land_rent"] == scenario["initial_land_rent"]*(1+scenario["rental_growth_rate"])**(i-1)
        for crop, key in (("grain", "initial_grain_other_cost"), ("cash_crop", "initial_cash_other_cost")):
            assert prices["other_cost_per_mu"][crop] == scenario[key]*(1+scenario["input_cost_growth_rate"])**(i-1)
        if portfolio and i == 1:
            decisions = old.read(path / (year+"decision_trace.json"))
            observed = {r["operator_id"]: r["previous_grain_share"] for r in decisions
                        if r["operator_type"] == "household" and r["previous_grain_share"] is not None}
            assert observed == shares, "initial shares did not enter actual household decisions"
        assert old.read(path / (year+"government_sar.json"))["action"] == 0
        weather = old.read(path / (year+"exogenous_shocks.json"))
        assert set(weather["stream_state_sha256"]) == {"b19_environment","b19_spatial_climate"}
        assert set(weather["spatial_maps"]) == {"current_hazard_map","current_favorable_map"}
        for name in (("price_inputs.json",) if portfolio else FILES) + ("exogenous_shocks.json",):
            compare(year + name)
    ok = all(r["match"] for r in checks.values())
    if portfolio:
        ok = ok and old.read(output/"design.json")["initial_parcels_sha256"] == old.read(
            paired_old/"design.json")["initial_parcels_sha256"]
    old.write(output/"audit.json", dict(seed=seed, rule=rule, ok=ok, baseline=str(baseline), checks=checks,
              scope="Bound weather streams/maps; no separate crop-price innovation in the crop market. "
                    "Realized market prices and individual yields are not common exogenous draws. "
                    "Archived paths do not store the separate price-input observer."))
    if not ok:
        raise RuntimeError(f"pairing/reproduction incomplete: {output}/audit.json")
    print("verified", output, flush=True)


if __name__ == "__main__":
    p = argparse.ArgumentParser(description=__doc__)
    p.add_argument("seed", type=int, choices=(103,109))
    p.add_argument("rule", choices=("old","new"))
    p.add_argument("--paired-old", type=Path)
    args = p.parse_args()
    run(args.seed, args.rule, paired_old=args.paired_old)

"""Build the NFPS sowing-area reference for the frozen job-8 households."""
import hashlib
import json
from pathlib import Path

import numpy as np
import pandas as pd


ROOT = Path(__file__).resolve().parents[1]
DIAG = ROOT / "results/diagnostics"
OUT = DIAG / "national_household_portfolio/reference"
RAW = ROOT / "校准用数据源/全国农村固定观察点/2009-2015household.csv"
MANIFEST = DIAG / "national_formal_revision_20261118/run_01/prepared_seed17_release1/preparation_manifest.json"


def prepare():
    job = json.loads(MANIFEST.read_text(encoding="utf-8"))["jobs"][8]
    cohort = Path(job["cohort"]).read_bytes()
    assert hashlib.sha256(cohort).hexdigest() == job["cohort_sha256"]
    farmers = pd.DataFrame(json.loads(cohort)["farmers"])
    cols = ["year", "hhid", "id", "hhnumbers", "sow_food", "sow_cash", "sow_total", "area_e"]
    raw = pd.read_csv(RAW, usecols=cols, dtype={"hhid": "string", "id": "string"},
                      keep_default_na=False, na_values=[""])
    origin = raw.iloc[farmers.source_row.to_numpy() - 2].reset_index(drop=True).copy()
    assert len(origin) == origin.hhid.nunique() == farmers.source_row.nunique() == farmers.cohort_agent_id.nunique() == 600
    assert origin.year.eq(2010).all() and origin.hhnumbers.eq(farmers.family_size).all()
    origin["actor_id"] = farmers.cohort_agent_id
    base = raw.loc[raw.year.eq(2010), "hhid"]
    ambiguous = base[base.duplicated(keep=False)]
    keys = origin.loc[~origin.hhid.isin(ambiguous), ["actor_id", "hhid", "id"]].dropna()
    references = []
    for year in (2010, 2011, 2012):
        if year == 2010:
            rows = origin
        else:
            current = raw.loc[raw.year.eq(year)].dropna(subset=["hhid", "id"])
            current = current.loc[~current.hhid.duplicated(keep=False)]
            rows = keys.merge(current, on=["hhid", "id"], validate="one_to_one")
        rows = rows.rename(columns={"sow_food": "grain_area", "sow_cash": "cash_area"}).copy()
        g, c = rows.grain_area, rows.cash_area
        for a in (g, c):
            assert (np.isfinite(a.dropna()) & a.dropna().ge(0)).all()
        rows["grain_share"] = g / (g + c).where((g + c).gt(0))
        physical = rows.area_e.notna() & rows.sow_total.notna()
        print(year, "matched, ambiguous_base, other_unmatched", len(rows),
              0 if year == 2010 else 600-len(keys), 0 if year == 2010 else len(keys)-len(rows),
              "grain_only, cash_only, both, neither, missing",
              *((g.gt(0) & c.eq(0)).sum(), (g.eq(0) & c.gt(0)).sum(),
                (g.gt(0) & c.gt(0)).sum(), (g.eq(0) & c.eq(0)).sum(), (g.isna() | c.isna()).sum()),
              "missing_grain, missing_cash", g.isna().sum(), c.isna().sum(),
              "share_valid", rows.grain_share.count(), "sown_gt_yearend, comparable",
              (physical & (rows.sow_total - rows.area_e).gt(0.00001)).sum(), physical.sum())
        references.append(rows[["year", "actor_id", "grain_area", "cash_area", "grain_share"]])
    ref = pd.concat(references, ignore_index=True).sort_values(["year", "actor_id"])
    assert not ref.duplicated(["year", "actor_id"]).any()
    valid = ref.grain_share.dropna()
    assert valid.between(0, 1).all()
    shares = ref.loc[ref.year.eq(2010)].set_index("actor_id").grain_share.dropna().to_dict()
    files = {"initial_grain_shares.json": json.dumps(shares, indent=2, allow_nan=False) + "\n",
             "reference_households.csv": ref.to_csv(index=False, lineterminator="\n")}
    OUT.mkdir(parents=True, exist_ok=True)
    for name, content in files.items():
        path = OUT / name
        if path.exists() and path.read_bytes() != content.encode("utf-8"):
            raise FileExistsError(f"Refusing to replace changed output: {path}")
    for name, content in files.items():
        path = OUT / name
        if not path.exists():
            with path.open("x", encoding="utf-8", newline="") as f:
                f.write(content)


if __name__ == "__main__":
    prepare()

"""Compare same-ID annual household crop areas before opening the policy comparison."""
import argparse
import json
from collections import Counter
from hashlib import sha256
from pathlib import Path

import pandas as pd


ROOT = Path(__file__).resolve().parents[1] / "results/diagnostics"
OUT = ROOT / "national_household_portfolio"


def areas(path, year):
    folder = path / f"year_{year-2010:02d}"
    snap = json.loads((folder / "actual_plot_use.json").read_text(encoding="utf-8"))
    harvests = json.loads((folder / "actual_harvests.json").read_text(encoding="utf-8"))
    if snap["year"] != year or snap["reconciliation"]["unreconciled_operators"]:
        raise ValueError(f"Unreconciled or misdated harvest: {folder}")
    # Plot allocations are additive at operator level only. No crop is assigned to an unidentified plot.
    df = pd.DataFrame(snap["patches"]).groupby("operator_id")[[
        "allocated_grain_area_mu", "allocated_economic_crop_area_mu"]].sum()
    df.columns = ["grain", "cash"]
    return df.reindex(harvests, fill_value=0)


def shares(a):
    total = a.grain + a.cash
    return a.grain.div(total.where(total > 0))


def stats(a):
    s = shares(a)
    return {"n":len(a), "positive_n":int(s.notna().sum()),
            "both":int(((a.grain > 0) & (a.cash > 0)).sum()),
            "both_min_below_1e5th_mu":int(((a.min(axis=1)>0) & (a.min(axis=1)<1e-5)).sum()),
            "mixed_rate":((a.grain > 0) & (a.cash > 0)).mean(),
            **dict(zip(("p25","p50","p75"),s.quantile([.25,.5,.75])))}


def decision_counts(path, year, ids):
    counts=Counter()
    with (path/f"year_{year-2010:02d}/actor_trace.jsonl").open(encoding="utf-8") as stream:
        for row in map(json.loads,stream):
            if row["actor_id"] not in ids:
                continue
            options=row["alternatives"] or []
            if not options:
                counts["no_current_options"]+=1
                continue
            mixed,other=[],[]
            offered=False
            for i,opt in enumerate(options):
                account=opt.get("expected_account")
                if account is None:
                    raise ValueError("Household candidate areas not observed")
                g,a=account["grain_area_mu"],account["area_mu"]
                interior=1e-8<g<a-1e-8
                offered=offered or interior
                if interior and row["selected_index"]==i:
                    counts["selected_interior"]+=1
                if opt["account_feasible"] and opt["service_available"] is not False:
                    (mixed if interior else other).append(opt["score_yuan"])
            if not mixed:
                counts["no_feasible_interior"]+=1
                counts["interior_offered_but_infeasible" if offered else "interior_not_offered"]+=1
            elif not other:
                counts["only_interior_feasible"]+=1
            elif max(mixed)>max(other)+1e-7:
                counts["interior_scores_higher"]+=1
            elif max(other)>max(mixed)+1e-7:
                counts["endpoint_or_exit_scores_higher"]+=1
            else:
                counts["score_tie"]+=1
    return {k:counts[k] for k in ("no_current_options","no_feasible_interior","interior_not_offered",
                  "interior_offered_but_infeasible","only_interior_feasible",
                  "interior_scores_higher","endpoint_or_exit_scores_higher","score_tie","selected_interior")}


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("year", type=int, choices=[2011,2012])
    parser.add_argument("--old", nargs=2, required=True, type=Path, metavar=("SEED103","SEED109"))
    parser.add_argument("--new", nargs=2, type=Path, metavar=("SEED103","SEED109"))
    args = parser.parse_args()
    survey = pd.read_csv(OUT / "reference/reference_households.csv").rename(
        columns={"grain_area":"grain","cash_area":"cash"})
    def actual(year):
        frame = survey[survey.year.eq(year)].set_index("actor_id",verify_integrity=True)[["grain","cash"]]
        return frame.dropna().loc[lambda a: (a>=0).all(axis=1)]
    y=args.year
    obs,prior=actual(y),actual(y-1)
    records=[]
    for seed,old,new in zip((103,109),args.old,args.new or (None,None)):
        for rule,path in (("old",old),("portfolio",new)):
            if path is None:
                continue
            sim=areas(path,y).reindex(obs.index)
            if new is not None:
                receipt=json.loads((path.parent/"audit.json").read_text(encoding="utf-8"))
                if not receipt["ok"] or receipt["seed"]!=seed or receipt["rule"]!=("new" if rule=="portfolio" else "old"):
                    raise ValueError(f"Unverified experiment: {path}")
            if sim.isna().any().any():
                raise ValueError(f"Missing actual harvest account: {path}, {y}")
            s,o=shares(sim),shares(obs)
            p=shares(prior) if y==2011 else shares(areas(path,y-1))
            common=(s.notna() & o.notna())
            motions=pd.concat([o,shares(prior),s,p],axis=1,join="inner").dropna()
            obs_step=(motions.iloc[:,0]-motions.iloc[:,1]).abs().mean()
            sim_step=(motions.iloc[:,2]-motions.iloc[:,3]).abs().mean()
            O,S=stats(obs),stats(sim)
            row=dict(year=y,seed=seed,rule=rule,common_positive_n=int(common.sum()),
                     shared_change_n=len(motions),obs_abs_change=obs_step,sim_abs_change=sim_step)
            if new is not None:
                row["run"]=path.parent.name
            for label,values in (("obs",O),("sim",S),("common_obs",stats(obs[common])),("common_sim",stats(sim[common]))):
                row.update({f"{label}_{k}":v for k,v in values.items()})
            row.update(mixed_error=abs(O["mixed_rate"]-S["mixed_rate"]),
                       quantile_mae=sum(abs(O[k]-S[k]) for k in ("p25","p50","p75"))/3,
                       change_error=abs(obs_step-sim_step))
            row.update(decision_counts(path,y,obs.index))
            records.append(row)
        if new is None:
            continue
        a,b=records[-2:]
        b["pass"]=bool(b["mixed_error"]<=min(.10,a["mixed_error"]+1e-12)
                       and b["quantile_mae"]<=min(.15,a["quantile_mae"]+1e-12)
                       and b["shared_change_n"]>0 and b["change_error"]<=.15)
    result=OUT/(f"behavior_{y}.csv" if args.new else f"baseline_{y}.csv")
    table=pd.DataFrame(records)
    table.to_csv(result,index=False,encoding="utf-8-sig")
    print(table.reindex(columns=["year","seed","rule","obs_n","obs_both","sim_both","mixed_error","quantile_mae","change_error","pass"]).to_string(index=False))
    if y==2012 and args.new:
        dev=pd.read_csv(OUT/"behavior_2011.csv")
        assert dev[["seed","run","rule"]].equals(table[["seed","run","rule"]])
        checks=pd.concat([dev,table]).query("rule == 'portfolio'")
        assert len(checks)==4
        unlocked=checks["pass"].eq(True).all()
        gate=dict(unlocked=bool(unlocked), rule="PROTOCOL.md §5",
                  failed=checks.loc[~checks["pass"].eq(True),["year","seed"]].to_dict("records"),
                  hashes={f:sha256((OUT/f).read_bytes()).hexdigest() for f in
                          ("PROTOCOL.md","behavior_2011.csv","behavior_2012.csv","reference/reference_households.csv")})
        (OUT/"behavior_gate.json").write_text(json.dumps(gate,indent=2,ensure_ascii=False),encoding="utf-8")
        if not unlocked:
            from scripts.audit_national_succession_transmission import performance
            annual=[]
            for s,old,new in zip((103,109),args.old,args.new):
                for r,path in (("old",old),("portfolio",new)):
                    ids={z["id"] for z in json.loads((path/"initial_households.json").read_text(encoding="utf-8"))}
                    for t in (1,2,3):
                        with (path/f"year_{t:02d}/actors.jsonl").open(encoding="utf-8") as f:
                            households=[z for z in map(json.loads,f) if z["id"] in ids]
                        assert len(households)==len({z["id"] for z in households})==len(ids)==600
                        accounts=[z["income_account"] for z in households]
                        assert all(a["step"]==t for a in accounts)
                        def mean(k):
                            return sum(a[k] for a in accounts)/600
                        out=performance(path,t)
                        pay=mean("support_at_harvest_yuan")+mean("verified_support_received_yuan")
                        wages=mean("off_farm_income_yuan")+mean("labor_sale_income_yuan")
                        rent=mean("rent_received_yuan")-mean("rent_paid_yuan")
                        operating=mean("sales_yuan")-sum(mean(k) for k in
                                   ("production_cost_yuan","service_cost_yuan","conversion_cost_yuan"))
                        other=mean("common_income_yuan")
                        residual=out["h0_cash_yuan"]-sum((pay,wages,rent,operating,other))
                        assert abs(residual)<1e-6,(path,t,residual)
                        annual.append(dict(seed=s,rule=r,year=2010+t,**out,h0_operating=operating,
                                           h0_wages=wages,h0_net_rent=rent,h0_support=pay,h0_common=other,
                                           account_residual=residual))
            base=pd.DataFrame(annual)
            base.to_csv(OUT/"baseline_accounts.csv",index=False,encoding="utf-8-sig")
            # Empty policy effects mean not evaluated, never a zero effect.
            status=pd.DataFrame([dict(seed=s,rule=r,policy=p,
                 status="behavior_only" if p=="base" else "locked_not_evaluated",
                 incremental_grain_t=None,incremental_h0_income_yuan=None,fiscal_cost_difference_yuan=None)
