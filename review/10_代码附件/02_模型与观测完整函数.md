class ActionIncomeAccount:
    action: str
    crop: str
    area_mu: float
    output_kg: float
    sales_yuan: float
    support_yuan: float
    rent_out_yuan: float
    off_farm_yuan: float
    common_income_yuan: float
    production_cost_yuan: float
    hired_labor_cost_yuan: float
    service_cost_yuan: float
    rent_in_yuan: float
    management_cost_yuan: float
    conversion_cost_yuan: float
    risk_penalty_yuan: float
    family_labor_days: float
    hired_labor_days: float
    off_farm_days: float
    service_area_mu: float
    working_capital_available: float
    hired_labor_available_days: float
    grain_area_mu: float = 0.0
    grain_output_kg: float = 0.0
    cash_output_kg: float = 0.0

    @property
    def cash_cost(self) -> float:
        return (self.production_cost_yuan + self.service_cost_yuan + self.rent_in_yuan
                + self.management_cost_yuan + self.conversion_cost_yuan)

    @property
    def working_capital_required(self) -> float:
        return self.cash_cost

    @property
    def net_income(self) -> float:
        return (self.sales_yuan + self.support_yuan + self.rent_out_yuan
                + self.off_farm_yuan + self.common_income_yuan - self.cash_cost)

    @property
    def score(self) -> float:
        return self.net_income - self.risk_penalty_yuan

    @property
    def constraint_reason(self) -> str:
        reasons = []
        if self.cash_cost > self.working_capital_available + 1e-8:
            reasons.append("working_capital")
        if self.hired_labor_days > self.hired_labor_available_days + 1e-8:
            reasons.append("hired_labor")
        return "|".join(reasons)

    @property
    def feasible(self) -> bool:
        return not self.constraint_reason

def choose_action_income(operator, accounts: tuple[ActionIncomeAccount, ...]) -> str:
    feasible = [account for account in accounts if account.feasible]
    if not feasible and getattr(operator.model, "initial_paid_lease_accounts", ()):
        terminal = next((a for a in accounts if a.action == "contract" and a.area_mu == 0), None)
        if terminal is not None and any(a.tenant is operator for a in operator.model.initial_paid_lease_accounts):
            return terminal.action
    if not feasible:
        raise RuntimeError(f"No feasible annual action for {operator.unique_id}")
    best = max(feasible, key=lambda account: account.score)
    prior_action = getattr(operator, "fixed_rule_previous_action", "maintain_grain")
    shares = getattr(operator, "household_candidate_grain_shares", None)
    if shares is not None and operator.household_previous_grain_share is not None:
        prior_action = next(key for key, share in shares.items() if share == operator.household_previous_grain_share)
    previous = next((a for a in feasible if a.action == prior_action), None)
    margin = fixed_rule_operator_inertia(operator)
    if previous is not None and best.score - previous.score <= margin:
        return previous.action
    return best.action

def organization_accounts(farmer: FarmerAgent, actions=("maintain_grain", "expand_cash", "exit")) -> tuple[OrganizationAccount, ...]:
    alternatives: list[OrganizationAccount] = []
    previous_mode = getattr(farmer, "p3_service_mode", None)
    # Keep the real rights-holder identity while the existing method previews costs.
    try:
        for action in actions:
            if action == "exit":
                alternatives.append(OrganizationAccount("transfer", farmer.expected_fixed_rule_account("exit"), True))
                continue
            for mode in ("self", "outsource"):
                setattr(farmer, "p3_service_mode", mode)
                account = farmer.expected_fixed_rule_account(action)
                alternatives.append(OrganizationAccount(mode, account,
                    mode == "self" or account.service_area_mu > 0))
    finally:
        setattr(farmer, "p3_service_mode", previous_mode)
    return tuple(alternatives)

def best_crop_accounts(alternatives: tuple[OrganizationAccount, ...]) -> tuple[ActionIncomeAccount, ...]:
    selected: list[ActionIncomeAccount] = []
    for action in dict.fromkeys(item.account.action for item in alternatives):
        choices = [item.account for item in alternatives if item.account.action == action and item.service_available]
        feasible = [account for account in choices if account.feasible]
        selected.append(max(feasible or choices, key=lambda account: account.score))
    return tuple(selected)

    def _calculate_deterministic_expected_yield(self, crop_type, area):
        """Mirror the structural yield function without climate/random shocks."""
        area = max(float(area or 0.0), 0.0)
        if area <= 0:
            return 0.0

        crop_name = str(getattr(crop_type, 'value', crop_type)).lower()

@contextmanager
def observe_actual_harvests(model):
    """Observe successful real settlements, restoring methods after each annual step."""
    model.revision_harvest_records = {}
    restored = []
    def observer(actor, original):
        def call(*args, **kwargs):
            identity = str(actor.unique_id)
            if identity in model.revision_harvest_records:
                raise RuntimeError("duplicate actual harvest settlement: " + identity)
            before = float(actor.capital)
            value = original(*args, **kwargs)
            model.revision_harvest_records[identity] = {
                "step": model.step_count, "capital_before_harvest": before,
                "capital_after_harvest": float(actor.capital),
                "grain_physical_kg": float(actor.grain_production),
                "cash_output_model_units": float(actor.cash_crop_production),
            }
            return value
        return call
    for actor in annual_actors(model).values():
        original = getattr(actor, "harvest_and_settle", None)
        if callable(original):
            local = "harvest_and_settle" in actor.__dict__
            restored.append((actor, original, local))
            actor.harvest_and_settle = observer(actor, original)
    try:
        yield
    finally:
        for actor, original, local in restored:
            if local:
                actor.harvest_and_settle = original
            else:
                del actor.harvest_and_settle

def run_episode(model, reference, output, seed, expected_sources, training, full_audit=False,
                observe_year=None):
    started = time.monotonic()
    output.mkdir()
    gov = model.government
    frozen = q_rows(gov.q_table)

    old.write(output / "q_before.json", frozen)
    old.write(output / "effective_config.json", model.config)
    initial = [old.household_row(f) for f in model.initial_household_cohort]
    old.write(output / "initial_households.json", initial)
    old.write(output / "initial_operators.json", old.operator_snapshot(model))
    initial_external = old.exogenous_snapshot(model)
    old.write(output / "initial_exogenous.json", initial_external)
    land = {str(p.get_patch_id()): float(p.area) for p in model.land_patches}
    opening, captured, shocks = {}, [], []

    def capture(current):

        if full_audit:
            old.record_year(current, directory, opening, seed, land)
            if "budget_area_menu" in current.config["policy"]["government_q_project"]:
                old.write(directory / "plot_production_status.json", [{
                    "id": str(p.get_patch_id()), "area_mu": p.area,

                    "crop": getattr(p.current_crop, "value", None),
                    "cultivated_area_mu": getattr(p, "preproduction_cultivated_area_mu", None),
                    "release_event_id": getattr(p, "production_succession_release_event_id", None),
                } for p in current.land_patches])
        else:
            directory.mkdir()
            old.write(directory / "annual_checks.json", old.check_annual_accounts(
                current, opening, land, old.operator_snapshot(current)))
            old.write(directory / "operator_snapshot.json", old.operator_snapshot(current))
            old.write(directory / "fiscal_account.json", current.government.current_subsidy_runtime)
        shock = old.exogenous_snapshot(current)
        if not full_audit:
            old.write(directory / "exogenous_shocks.json", shock)
        shocks.append(shock)
        if observe_year is not None:
            observe_year(current, directory)
        captured.append(current.step_count)

    model._completed_production_year_snapshot_hook = capture
    years = model.config["policy"]["government_q_project"]["episode_end_step"]
    try:
        for index in range(years):
            check_sources(expected_sources)
            opening = {key: float(a.capital) for key, a in old.annual_actors(model).items()}
            old.apply_annual_exogenous_path(model, index)
            old.advance_program(model, index, reference)
            with old.observe_actual_harvests(model):
                model.step()
            if captured != list(range(1, index + 2)) or len(gov.government_q_sar) != index + 1:

                raise RuntimeError("evaluation changed frozen Q")

                  "initial_exogenous_sha256": fingerprint(initial_external),

                  "frozen_q_unchanged": None if training else q_rows(gov.q_table) == frozen}
        result.update(reward_summary(gov.government_q_sar, gov.discount_factor))
        old.write(output / "COMPLETED.json", result)
        return result
    except BaseException as error:
        old.write(output / "FAILED.json", {"error": str(error)})
        raise
    finally:
